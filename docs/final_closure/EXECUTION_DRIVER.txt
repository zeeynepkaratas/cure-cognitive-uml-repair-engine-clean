def now():return datetime.datetime.now(datetime.timezone(datetime.timedelta(hours=3))).isoformat()
def clean(text):
    text=text.replace(str(ROOT),'<repo>').replace(str(WORK),'<work>')
    return re.sub(r'[A-Za-z]:[\\/]+Users[\\/]+[^\\/\s]+','<local-user>',text)
def save():

def run(name,args,cwd=ROOT,timeout=600,required=True):
    started=now();t=time.monotonic()
    try:
        r=subprocess.run(args,cwd=cwd,env=os.environ.copy(),capture_output=True,text=True,encoding='utf8',errors='replace',timeout=timeout)
        exitcode=r.returncode;stdout=r.stdout;stderr=r.stderr
    except (OSError,subprocess.TimeoutExpired) as exc:
        exitcode=None;stdout='';stderr=type(exc).__name__+': execution did not complete'
    rec={'name':name,'command':args,'cwd':'frontend' if cwd==ROOT/'frontend' else '.','started_at':started,'finished_at':now(),'timezone':'Europe/Istanbul','duration_seconds':time.monotonic()-t,'exit_code':exitcode,'status':'PASS' if exitcode==0 else 'FAIL' if exitcode is not None else 'INCONCLUSIVE','stdout':clean(stdout),'stderr':clean(stderr)}
    records.append(rec);save();print(json.dumps({'check':name,'exit_code':exitcode,'status':rec['status']}),flush=True)
    if required and exitcode!=0:raise SystemExit(1)
    return exitcode
